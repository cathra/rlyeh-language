```rust
//! Zeta 语言语法分析器
//!
//! 将 Token 流转换为抽象语法树（AST）。

#![warn(missing_docs)]
#![warn(unsafe_code)]

use zeta_lexer::{Lexer, LocatedToken, Token, Span};
use zeta_ast::*;
use thiserror::Error;

/// 语法分析器
pub struct Parser<'src> {
    tokens: std::iter::Peekable<std::vec::IntoIter<LocatedToken>>,
    source: &'src str,
}

/// 语法错误
#[derive(Debug, Error)]
pub enum ParseError {
    #[error("unexpected token: expected {expected}, found {found} at {line}:{col}")]
    UnexpectedToken {
        expected: String,
        found: String,
        line: usize,
        col: usize,
    },
    
    #[error("unexpected end of file")]
    UnexpectedEof,
    
    #[error("invalid comparison chain at {line}:{col}: {reason}")]
    InvalidComparisonChain {
        reason: String,
        line: usize,
        col: usize,
    },
    
    #[error("invalid region syntax at {line}:{col}: {reason}")]
    InvalidRegion {
        reason: String,
        line: usize,
        col: usize,
    },
    
    #[error("duplicated modifier at {line}:{col}")]
    DuplicatedModifier {
        modifier: String,
        line: usize,
        col: usize,
    },
}

impl<'src> Parser<'src> {
    /// 创建解析器
    pub fn new(source: &'src str) -> Result<Self, ParseError> {
        let mut lexer = Lexer::new(source);
        let tokens = lexer.tokenize()?;
        Ok(Self {
            tokens: tokens.into_iter().peekable(),
            source,
        })
    }
    
    /// 解析整个程序
```

```rust
// crates/zeta-ast/src/lib.rs

//! Zeta 抽象语法树定义

#![warn(missing_docs)]

/// 完整程序
pub struct AstProgram {
    pub items: Vec<AstItem>,
}

/// 顶层项
pub enum AstItem {
    FnDecl(Box<AstFnDecl>),
    StructDecl(Box<AstStructDecl>),
    EnumDecl(Box<AstEnumDecl>),
    TraitDecl(Box<AstTraitDecl>),
    ImplBlock(Box<AstImplBlock>),
    ModDecl(Box<AstModDecl>),
    UseDecl(Box<AstUseDecl>),
    ConstDecl(Box<AstConstDecl>),
    ActorDecl(Box<AstActorDecl>),
    MacroDecl(Box<AstMacroDecl>),
    Statement(Box<AstStmt>),
}

/// 表达式（带 span 信息）
pub struct AstExpr {
    pub kind: Box<ExprKind>,
    pub span: Span,
}

/// 表达式种类
pub enum ExprKind {
    // 字面量
    IntLiteral(i128),
    FloatLiteral(f64),
    StringLiteral(String),
    CharLiteral(char),
    BoolLiteral(bool),
    TimeLiteral { hour: u8, minute: u8, is_pm: bool },
    
    // 标识符与路径
    Ident(String),
    Path(Vec<String>),
    
    // 二元运算
    Binary {
        op: BinaryOp,
        left: AstExpr,
        right: AstExpr,
    },
    
    // 一元运算
    Unary {
        op: UnaryOp,
        operand: AstExpr,
    },
    
    // 比较链（核心特性）
    ComparisonChain {
        elements: Vec<AstExpr>,
        operators: Vec<CompareOp>,
    },
    
    // 集合判断（in 表达式）
    InSet {
        value: AstExpr,
        set: Vec<AstExpr>,
    },
    
    // between 表达式
    Between {
        value: AstExpr,
        lower: AstExpr,
        upper: AstExpr,
        lower_exclusive: bool,
        upper_exclusive: bool,
    },
    
    // 赋值
    Assign {
        target: AstExpr,
        op: AssignOp,
        value: AstExpr,
    },
    
    // 控制流
    If {
        cond: AstExpr,
        then_block: AstBlock,
        else_block: Option<AstBlock>,
    },
    
    Match {
        expr: AstExpr,
        arms: Vec<MatchArm>,
    },
    
    For {
        pattern: AstPattern,
        iterator: AstExpr,
        body: AstBlock,
    },
    
    While {
        cond: AstExpr,
        body: AstBlock,
    },
    
    Loop {
        body: AstBlock,
    },
    
    // 区域
    Region {
        name: Option<String>,
        options: RegionOptions,
        body: AstBlock,
    },
    
    // Transfer
    Transfer {
        expr: AstExpr,
        region: String,
    },
    
    // 函数调用
    Call {
        callee: AstExpr,
        args: Vec<AstExpr>,
    },
    
    // 方法调用
    MethodCall {
        receiver: AstExpr,
        method: String,
        args: Vec<AstExpr>,
    },
    
    // 字段访问
    FieldAccess {
        expr: AstExpr,
        field: String,
    },
    
    // 索引
    Index {
        expr: AstExpr,
        index: AstExpr,
    },
    
    // 闭包
    Closure {
        params: Vec<AstPattern>,
        body: AstExpr,
        capture: CaptureMode,
    },
    
    // 类型转换
    Cast {
        expr: AstExpr,
        target_type: AstType,
    },
    
    // 块
    Block(AstBlock),
    
    // 返回
    Return(Option<AstExpr>),
    
    // break / continue
    Break(Option<AstExpr>),
    Continue,
    
    // Actor 消息发送
    Send {
        actor: AstExpr,
        method: String,
        args: Vec<AstExpr>,
    },
}

/// 比较运算符
pub enum CompareOp {
    Lt, Le, Gt, Ge, Eq, Ne,
}

/// 二元运算符
pub enum BinaryOp {
    Add, Sub, Mul, Div, Mod,
    BitAnd, BitOr, BitXor, Shl, Shr,
    And, Or,
}

/// 一元运算符
pub enum UnaryOp {
    Neg, Not, Deref, AddrOf, AddrOfMut,
}

/// 赋值运算符
pub enum AssignOp {
    Assign, AddAssign, SubAssign, MulAssign, DivAssign,
}

/// 闭包捕获模式
pub enum CaptureMode {
    Move,  // move || { ... }
    Borrow, // || { ... } (默认)
}

/// 区域选项
pub struct RegionOptions {
    pub size: Option<usize>,
    pub allow_growth: bool,
    pub growth_factor: Option<f64>,
    pub adaptive: bool,
    pub exact: bool,
}

/// 代码块
pub struct AstBlock {
    pub stmts: Vec<AstStmt>,
    pub final_expr: Option<AstExpr>,  // 块的最后表达式
    pub span: Span,
}

/// 语句
pub enum AstStmt {
    Let {
        pattern: AstPattern,
        type_anno: Option<AstType>,
        init: AstExpr,
        mutable: bool,
    },
    Expr(AstExpr),
    Item(AstItem),
    Semi(AstExpr),  // 表达式 + 分号
}

/// 模式
pub enum AstPattern {
    Ident(String),
    Wildcard,
    Literal(LiteralValue),
    Tuple(Vec<AstPattern>),
    Struct(String, Vec<(String, AstPattern)>),
    Enum(String, Vec<AstPattern>),
    Range { lower: AstExpr, upper: AstExpr, inclusive: bool },
    Ref(Box<AstPattern>, bool),  // ref mut? pat
}
```

## 测试用例

```rust
// crates/zeta-parser/tests/parser_test.rs

use zeta_parser::{Parser, AstProgram};
use zeta_lexer::Lexer;

fn parse(source: &str) -> AstProgram {
    let mut parser = Parser::new(source).unwrap();
    parser.parse_program().unwrap()
}

#[test]
fn test_basic_let() {
    let ast = parse("let x = 42;");
    assert_eq!(ast.items.len(), 1);
}

#[test]
fn test_comparison_chain_forward() {
    let ast = parse("if 0 < x < 10 {}");
    // 验证 AST 中包含 ComparisonChain
    // elements: [0, x, 10]
    // operators: [Lt, Lt]
}

#[test]
fn test_comparison_chain_reverse() {
    let ast = parse("if 0 > x > 10 {}");
    // elements: [0, x, 10]
    // operators: [Gt, Gt]  → 语义分析阶段解释为区间外
}

#[test]
fn test_in_expression() {
    let ast = parse("if x in (1, 3, 5) {}");
    // 验证 InSet { value: x, set: [1, 3, 5] }
}

#[test]
fn test_not_in_expression() {
    let ast = parse("if x not in (1, 2, 3) {}");
    // 验证 InSet 的取反
}

#[test]
fn test_between_expression() {
    let ast = parse("if x between 0 and 10 {}");
    // 验证 Between { lower: 0, upper: 10, ... }

#[test]

    let ast = parse("region 'r { let x = 42 in 'r; }");

}

#[test]
fn test_region_with_options() {
```
