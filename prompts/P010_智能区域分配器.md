## 模块 1：静态大小推断 (static_sizer)

```rust
// crates/zeta-region-alloc/src/static_sizer.rs

#![warn(missing_docs)]

use std::collections::HashMap;

/// 编译期区域大小推断结果
#[derive(Debug, Clone)]
pub enum SizeEstimate {
    /// 精确大小（所有分配在编译期可知）
    Exact(usize),
    /// 上界（有循环，但循环次数可知）
    UpperBound(usize),
    /// 区间（最小和最大可能大小）
    Range { min: usize, max: usize },
    /// 无法推断（完全动态）
    Unknown,
}

/// 区域内单个分配点
#[derive(Debug, Clone)]
pub struct AllocSite {
    /// 分配点 ID（用于去重）
    pub id: u64,
    /// 类型大小
    pub type_size: usize,
    /// 对齐要求
    pub alignment: usize,
    /// 该分配点的执行次数估计
    pub frequency: ExecFrequency,
}

/// 执行频率估计
#[derive(Debug, Clone)]
pub enum ExecFrequency {
    /// 精确一次
    Once,
    /// 已知次数（编译期常量）
    Constant(usize),
    /// 范围
    Range { min: usize, max: usize },
    /// 未知
    Unknown,
}

/// 静态大小推断器
pub struct StaticSizer {
    /// 区域内所有分配点
    alloc_sites: Vec<AllocSite>,
    /// 条件分支信息
    branches: Vec<BranchInfo>,
}

/// 条件分支信息
#[derive(Debug, Clone)]
pub struct BranchInfo {
    /// 分支 ID
    pub id: u64,
    /// 各分支的分配点集合
    pub branches: Vec<Vec<u64>>,  // 每个分支包含的 alloc_site ID
}

impl StaticSizer {
    pub fn new() -> Self {
        Self {
            alloc_sites: vec![],
            branches: vec![],
        }
    }
    
    /// 添加一个分配点
    pub fn add_alloc_site(&mut self, site: AllocSite) {
        self.alloc_sites.push(site);
    }
    
    /// 添加条件分支信息
    pub fn add_branch(&mut self, branch: BranchInfo) {
        self.branches.push(branch);
    }
    
    /// 计算区域总大小估计
    /// 
    /// 算法：
    /// 1. 无分支：所有 alloc_site 的 size × frequency 之和
    /// 2. 有分支：取所有分支路径中的最大值
    /// 3. 循环：frequency 为 Constant(n) 时，size × n
    pub fn estimate(&self) -> SizeEstimate {
        if self.branches.is_empty() {
            self.estimate_linear()
        } else {
            self.estimate_with_branches()
        }
    }
    
    fn estimate_linear(&self) -> SizeEstimate {
        let mut total_min = 0usize;
        let mut total_max = 0usize;
        let mut exact = true;
        let mut has_bounded_loop = false;
        
        for site in &self.alloc_sites {
            let (min, max) = match &site.frequency {
                ExecFrequency::Once => (site.type_size, site.type_size),
                ExecFrequency::Constant(n) => {
                    exact = false;
                    has_bounded_loop = true;
                    (site.type_size * n, site.type_size * n)
                }
                ExecFrequency::Range { min, max } => {
                    exact = false;
                    (site.type_size * min, site.type_size * max)
                }
                ExecFrequency::Unknown => {
                    return SizeEstimate::Unknown;
                }
            };
            
            total_min += min;
            total_max += max;
        }
        
        if exact {
            SizeEstimate::Exact(total_max)
        } else if has_bounded_loop {
            // 有界循环：循环次数可知，报告上界而非精确值
            SizeEstimate::UpperBound(total_max)
        } else if total_min == total_max {
            SizeEstimate::Exact(total_max)
        } else {
            SizeEstimate::Range { min: total_min, max: total_max }
        }
    }
    
    fn estimate_with_branches(&self) -> SizeEstimate {
        // 为每条分支路径计算总大小
        // 取最大值作为 max，取最小值作为 min
        
        let mut path_sums: Vec<usize> = vec![0];
        
        // 简化：遍历所有分支组合
        // 实际实现中需要构建控制流图
        
        for branch in &self.branches {
            let mut new_sums = vec![];
            
            for current_sum in &path_sums {
                for branch_sites in &branch.branches {
                    let mut sum = *current_sum;
                    for site_id in branch_sites {
                        if let Some(site) = self.alloc_sites.iter().find(|s| s.id == *site_id) {
                            sum += match &site.frequency {
                                ExecFrequency::Once => site.type_size,
                                ExecFrequency::Constant(n) => site.type_size * n,
                                _ => site.type_size, // 保守估计
                            };
                        }
                    }
                    new_sums.push(sum);
                }
            }
            
            path_sums = new_sums;
        }
        
        if path_sums.is_empty() {
            return SizeEstimate::Unknown;
        }
        
        let min = *path_sums.iter().min().unwrap();
        let max = *path_sums.iter().max().unwrap();
        
        if min == max {
            SizeEstimate::Exact(max)
        } else {
            SizeEstimate::Range { min, max }
        }
    }
    
    /// 考虑对齐后的总大小
    pub fn estimate_aligned(&self, default_align: usize) -> SizeEstimate {
        let estimate = self.estimate();
        
        match estimate {
            SizeEstimate::Exact(n) => {
                let aligned = (n + default_align - 1) & !(default_align - 1);
                SizeEstimate::Exact(aligned)
            }
            SizeEstimate::Range { min, max } => {
                let min_aligned = (min + default_align - 1) & !(default_align - 1);
                let max_aligned = (max + default_align - 1) & !(default_align - 1);
                SizeEstimate::Range { min: min_aligned, max: max_aligned }
            }
            SizeEstimate::UpperBound(n) => {
                let aligned = (n + default_align - 1) & !(default_align - 1);
                SizeEstimate::UpperBound(aligned)
            }
            SizeEstimate::Unknown => SizeEstimate::Unknown,
        }
    }
}
```

```rust
// crates/zeta-region-alloc/src/pgo_advisor.rs

#![warn(missing_docs)]

use serde::{Deserialize, Serialize};
use std::collections::HashMap;

/// 当前 PGO 数据格式版本
pub const CURRENT_PGO_VERSION: u32 = 1;

/// PGO 配置文件
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PgoData {
    pub version: u32,
    pub regions: HashMap<String, RegionProfile>,
}

/// 单个区域的 profile 数据
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RegionProfile {
    /// 调用次数
    pub call_count: u64,
    /// 大小统计
    pub size_stats: SizeStats,
    /// 扩容统计
    pub growth_stats: GrowthStats,
    /// 每次调用的分配采样
    pub samples: Vec<usize>,
}

/// 大小统计
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SizeStats {
    pub min: usize,
    pub max: usize,
    pub avg: f64,
    pub p50: usize,
    pub p95: usize,
    pub p99: usize,
}

/// 扩容统计
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GrowthStats {
    /// 平均扩容次数
    pub avg_growth_count: f64,
    /// 平均扩容因子
    pub avg_growth_factor: f64,
    /// 最大扩容因子
    pub max_growth_factor: f64,
}

/// PGO 顾问
pub struct PgoAdvisor {
    data: Option<PgoData>,
}

impl PgoAdvisor {
    pub fn new() -> Self {
        Self { data: None }
    }
    
    /// 从内存中的 PGO 数据创建
    pub fn from_data(data: PgoData) -> Self {
        Self { data: Some(data) }
    }
    
    /// 从文件加载 PGO 数据
    pub fn load_from_file(path: &std::path::Path) -> Result<Self, PgoError> {
        let json = std::fs::read_to_string(path)?;
        let data: PgoData = serde_json::from_str(&json)?;
        if data.version != CURRENT_PGO_VERSION {
            return Err(PgoError::VersionMismatch {
                expected: CURRENT_PGO_VERSION,
                found: data.version,
            });
        }
        Ok(Self { data: Some(data) })
    }
    
    /// 保存 PGO 数据到文件
    pub fn save_to_file(&self, path: &std::path::Path) -> Result<(), PgoError> {
        if let Some(data) = &self.data {
            let json = serde_json::to_string_pretty(data)?;
            std::fs::write(path, json)?;
        }
        Ok(())
    }
    
    /// 为指定区域推荐初始大小
    /// 
    /// 策略：
    /// - 有 PGO 数据 → 使用 P95（覆盖 95% 的调用）
    /// - 无 PGO 数据 → 返回 None（交给其他策略）
    pub fn recommend_size(&self, region_id: &str) -> Option<usize> {
        let data = self.data.as_ref()?;
        let profile = data.regions.get(region_id)?;
        
        // 使用 P95 值
        // 这样 95% 的调用不需要扩容
        let recommended = profile.size_stats.p95;
        
        // 加 10% 安全余量
        let with_margin = (recommended as f64 * 1.1) as usize;
        
        Some(with_margin)
    }
    
    /// 推荐扩容策略参数
    pub fn recommend_strategy(&self, region_id: &str) -> Option<StrategyParams> {
        let data = self.data.as_ref()?;
        let profile = data.regions.get(region_id)?;
        
        Some(StrategyParams {
            growth_factor: profile.growth_stats.avg_growth_factor.max(1.5),
            min_size: profile.size_stats.p50,
            max_size: profile.size_stats.max * 4, // 4x 最大历史值
        })
    }
    
    /// 检查 PGO 数据是否足够
    pub fn has_sufficient_data(&self, region_id: &str) -> bool {
        if let Some(data) = &self.data {
            if let Some(profile) = data.regions.get(region_id) {
                return profile.call_count >= 30; // 至少 30 次采样
            }
        }
        false
    }
}

/// 策略参数
#[derive(Debug, Clone)]
pub struct StrategyParams {
    pub growth_factor: f64,
    pub min_size: usize,
    pub max_size: usize,
}

/// PGO 错误
#[derive(Debug, thiserror::Error)]
pub enum PgoError {
    #[error("failed to read PGO file: {0}")]
    Io(#[from] std::io::Error),
    
    #[error("failed to parse PGO JSON: {0}")]
    Serde(#[from] serde_json::Error),
    
    #[error("invalid PGO data: {0}")]
    InvalidData(String),
    
    #[error("PGO version mismatch: expected {expected}, found {found}")]
    VersionMismatch { expected: u32, found: u32 },
}
```

```rust
// crates/zeta-region-alloc/src/size_advisor.rs

#![warn(missing_docs)]

use crate::static_sizer::{SizeEstimate, StaticSizer};
use crate::pgo_advisor::{PgoAdvisor, StrategyParams};

/// 最终决策结果
#[derive(Debug, Clone)]
pub struct AllocatorPlan {
    /// 推荐初始大小
    pub initial_size: usize,
    /// 推荐最小大小（不允许缩到更小）
    pub min_size: usize,
    /// 推荐最大大小（超过则报错或拒绝）
    pub max_size: usize,
    /// 扩容因子
    pub growth_factor: f64,
    /// 决策来源（用于调试）
    pub decision_source: DecisionSource,
}

/// 决策来源
#[derive(Debug, Clone)]
pub enum DecisionSource {
    /// 完全静态推断
    StaticExact(usize),
    /// 静态上界
    StaticUpperBound(usize),
    /// PGO 数据
    Pgo(usize),
    /// 自适应（无数据）
    AdaptiveDefault,
    /// 用户指定
    UserSpecified(usize),
}

/// 综合决策器
pub struct SizeAdvisor {
    static_sizer: Option<StaticSizer>,
    pgo: Option<PgoAdvisor>,
    user_hint: Option<usize>,
    region_id: String,
}

impl SizeAdvisor {
    pub fn new(region_id: String) -> Self {
        Self {
            static_sizer: None,
            pgo: None,
            user_hint: None,
            region_id,
        }
    }
    
    /// 设置静态推断器
    pub fn with_static(mut self, sizer: StaticSizer) -> Self {
        self.static_sizer = Some(sizer);
        self
    }
    
    /// 设置 PGO 顾问
    pub fn with_pgo(mut self, pgo: PgoAdvisor) -> Self {
        self.pgo = Some(pgo);
        self
    }
    
    /// 设置用户提示
    pub fn with_user_hint(mut self, size: usize) -> Self {
        self.user_hint = Some(size);
        self
    }
    
    /// 做出最终决策
    /// 
    /// 优先级：
    /// 1. 用户精确指定（exact）→ 直接使用
    /// 2. 静态精确推断 → 使用静态值
    /// 3. PGO 数据充分 → 使用 PGO P95
    /// 4. 静态上界 → 使用上界
    /// 5. 用户提示 → 使用提示值
    /// 6. 兜底 → 自适应默认值（4KB）
    pub fn decide(self) -> AllocatorPlan {
        // 1. 用户精确指定
        if let Some(hint) = self.user_hint {
            return AllocatorPlan {
                initial_size: hint,
                min_size: hint,
                max_size: hint * 4,
                growth_factor: 1.0, // 不需要扩容
                decision_source: DecisionSource::UserSpecified(hint),
            };
        }
        
        // 2. 静态精确推断
        if let Some(sizer) = &self.static_sizer {
            let estimate = sizer.estimate();
            if let SizeEstimate::Exact(n) = estimate {
                let with_margin = (n as f64 * 1.1) as usize; // 10% 余量
                return AllocatorPlan {
                    initial_size: with_margin,
                    min_size: with_margin,
                    max_size: with_margin * 2,
                    growth_factor: 1.5, // 保险起见允许少量扩容
                    decision_source: DecisionSource::StaticExact(n),
                };
            }
        }
        
        // 3. PGO 数据
        if let Some(pgo) = &self.pgo {
            if pgo.has_sufficient_data(&self.region_id) {
                if let Some(size) = pgo.recommend_size(&self.region_id) {
                    let strategy = pgo.recommend_strategy(&self.region_id);
                    let (gf, max) = strategy.map_or((2.0, size * 4), |s| {
                        (s.growth_factor, s.max_size)
                    });
                    
                    return AllocatorPlan {
                        initial_size: size,
                        min_size: size / 2,
                        max_size: max,
                        growth_factor: gf,
                        decision_source: DecisionSource::Pgo(size),
                    };
                }
            }
        }
        
        // 4. 静态上界
        if let Some(sizer) = &self.static_sizer {
            let estimate = sizer.estimate();
            let size = match estimate {
                SizeEstimate::UpperBound(n) => Some(n),
                SizeEstimate::Range { max, .. } => Some(max),
                _ => None,
            };
            if let Some(size) = size {
                return AllocatorPlan {
                    initial_size: size,
                    min_size: size / 2,
                    max_size: size * 4,
                    growth_factor: 2.0,
                    decision_source: DecisionSource::StaticUpperBound(size),
                };
            }
        }
        
        // 5. 兜底：自适应默认值
        let default = 4 * 1024; // 4KB
        AllocatorPlan {
            initial_size: default,
            min_size: 1024,     // 1KB
            max_size: 1024 * 1024 * 64, // 64MB
            growth_factor: 2.0,
            decision_source: DecisionSource::AdaptiveDefault,
        }
    }
}
```

## 4：自适应 Bump Allocator（完整版）

```rust
// crates/zeta-region-alloc/src/allocator.rs

#![warn(missing_docs)]

use std::alloc::{alloc, dealloc, Layout};
use std::ptr::NonNull;
use std::time::Instant;
use std::collections::VecDeque;

use crate::size_advisor::{AllocatorPlan, SizeAdvisor, DecisionSource};
use crate::stats::{RegionStats, AllocEvent, GrowthEvent};

/// 单个内存块
struct MemoryBlock {
    start: NonNull<u8>,
    end: NonNull<u8>,
    current: NonNull<u8>,
    size: usize,
    /// 碎片统计
    fragmentation: usize,
}

impl MemoryBlock {
    fn new(size: usize) -> Result<Self, AllocError> {
        let layout = Layout::from_size_align(size, 16).unwrap();
        let ptr = unsafe { alloc(layout) };
        if ptr.is_null() {
            return Err(AllocError::OutOfMemory);
        }
        
        let start = NonNull::new(ptr).unwrap();
        Ok(Self {
            start,
            end: unsafe { NonNull::new_unchecked(ptr.add(size)) },
            current: start,
            size,
            fragmentation: 0,
        })
    }
    
    #[inline]
    fn allocate(&mut self, size: usize, align: usize) -> Option<NonNull<u8>> {
        let current_addr = self.current.as_ptr() as usize;
        let aligned = (current_addr + align - 1) & !(align - 1);
        
        let new_current = aligned + size;
        if new_current > self.end.as_ptr() as usize {
            // 记录碎片
            self.fragmentation += self.end.as_ptr() as usize - current_addr;
            return None;
        }
        
        let ptr = unsafe { NonNull::new_unchecked(aligned as *mut u8) };
        self.current = unsafe { NonNull::new_unchecked(new_current as *mut u8) };
        Some(ptr)
    }
    
    #[inline]
    fn reset(&mut self) {
        self.current = self.start;
    }
    
    fn used(&self) -> usize {
        self.current.as_ptr() as usize - self.start.as_ptr() as usize
    }
    
    fn remaining(&self) -> usize {
        self.end.as_ptr() as usize - self.current.as_ptr() as usize
    }
}

impl Drop for MemoryBlock {
    fn drop(&mut self) {
        let layout = Layout::from_size_align(self.size, 16).unwrap();
        unsafe { dealloc(self.start.as_ptr(), layout) };
    }
}

/// 析构条目
struct DestructorEntry {
    ptr: NonNull<u8>,
    drop_fn: unsafe fn(NonNull<u8>),
    size: usize,
    align: usize,
}

/// 智能区域分配器
pub struct SmartRegion {
    /// 区域名称
    name: String,
    /// 内存块链表
    blocks: Vec<MemoryBlock>,
    /// 当前活跃块索引
    current_block: usize,
    /// 析构函数列表
    destructors: Vec<DestructorEntry>,
    /// 已转移对象
    transferred: Vec<NonNull<u8>>,
    /// 分配计划
    plan: AllocatorPlan,
    /// 统计
    stats: RegionStats,
    /// 扩容历史
    growth_history: Vec<GrowthEvent>,
    /// 分配采样
    allocation_samples: Vec<usize>,
    /// EWMA 状态
    ewma_state: EwmaState,
    /// 创建时间
    created_at: Instant,
}

/// EWMA 指数加权移动平均
///
/// `pub`：bench 与外部工具需要直接访问。
pub struct EwmaState {
    /// 平滑因子
    alpha: f64,
    /// 当前 EWMA 值
    value: f64,
    /// 是否已初始化
    initialized: bool,
}

impl EwmaState {
    pub fn new(alpha: f64) -> Self {
        Self {
            alpha,
            value: 0.0,
            initialized: false,
        }
    }
    
    /// 更新 EWMA
    pub fn update(&mut self, sample: f64) {
        if !self.initialized {
            self.value = sample;
            self.initialized = true;
        } else {
            self.value = self.alpha * sample + (1.0 - self.alpha) * self.value;
        }
    }
    
    pub fn current(&self) -> f64 {
        self.value
    }
}

/// 分配错误
#[derive(Debug, thiserror::Error)]
pub enum AllocError {
    #[error("out of memory")]
    OutOfMemory,
    
    #[error("allocation too large: requested {requested}, max {max}")]
    TooLarge { requested: usize, max: usize },
    
    #[error("region is frozen (transferred out)")]
    Frozen,
}

impl SmartRegion {
    /// 创建智能区域
    /// 
    /// 通过 SizeAdvisor 自动决定初始大小
    pub fn new(name: String, advisor: SizeAdvisor) -> Result<Self, AllocError> {
        let plan = advisor.decide();
        let initial_size = plan.initial_size.max(1024); // 至少 1KB
        
        let block = MemoryBlock::new(initial_size)?;
        
        let stats = RegionStats {
            region_name: name.clone(),
            created_at: std::time::SystemTime::now()
                .duration_since(std::time::UNIX_EPOCH)
                .map(|d| d.as_secs().to_string())
                .unwrap_or_default(),
            total_allocated: 0,
            total_deallocated: 0,
            allocation_count: 0,
            growth_count: 0,
            destructor_calls: 0,
            peak_usage: 0,
            fragmentation: 0,
            decision_source: format!("{:?}", plan.decision_source),
        };
        
        Ok(Self {
            name,
            blocks: vec![block],
            current_block: 0,
            destructors: vec![],
            transferred: vec![],
            plan,
            stats,
            growth_history: vec![],
            allocation_samples: vec![],
            ewma_state: EwmaState::new(0.3),
            created_at: Instant::now(),
        })
    }
    
    /// 使用用户指定大小创建
    pub fn with_size(name: String, size: usize) -> Result<Self, AllocError> {
        let advisor = SizeAdvisor::new(name.clone())
            .with_user_hint(size);
        Self::new(name, advisor)
    }
    
    /// 使用 PGO 数据创建
    pub fn with_pgo(name: String, pgo_data: crate::profile::PgoData) -> Result<Self, AllocError> {
        let pgo_advisor = crate::pgo_advisor::PgoAdvisor::new();
        // ... 设置 PGO 数据
        
        let advisor = SizeAdvisor::new(name.clone())
            .with_pgo(pgo_advisor);
        Self::new(name, advisor)
    }
    
    /// 在区域内分配对象
    #[inline]
    pub fn allocate<T>(&mut self, value: T) -> Result<&mut T, AllocError> {
        let size = std::mem::size_of::<T>();
        let align = std::mem::align_of::<T>();
        
        // 尝试在当前块分配
        let ptr = match self.blocks[self.current_block].allocate(size, align) {
            Some(ptr) => ptr,
            None => {
                // 需要扩容
                self.grow(size, align)?;
                self.blocks[self.current_block].allocate(size, align)
                    .ok_or(AllocError::OutOfMemory)?
            }
        };
        
        // 写入值
        unsafe {
            std::ptr::write(ptr.as_ptr() as *mut T, value);
        }
        
        // 如果 T 需要析构，注册析构函数
        if needs_drop::<T>() {
            self.destructors.push(DestructorEntry {
                ptr,
                drop_fn: drop_in_place::<T>,
                size,
                align,
            });
        }
        
        // 更新统计
        self.stats.total_allocated += size;
        self.stats.allocation_count += 1;
        self.allocation_samples.push(size);
        
        let used = self.current_used();
        if used > self.stats.peak_usage {
            self.stats.peak_usage = used;
        }
        
        Ok(unsafe { &mut *(ptr.as_ptr() as *mut T) })
    }
    
    /// 扩容
    fn grow(&mut self, needed: usize, align: usize) -> Result<(), AllocError> {
        let current_size = self.blocks[self.current_block].size;
        
        // 使用 EWMA 预测下次需要的大小
        let predicted = self.ewma_state.current();
        let ewma_suggested = (predicted * 1.5) as usize; // 50% 余量
        
        // 根据决策来源选择策略
        let new_size = match self.plan.decision_source {
            DecisionSource::StaticExact(_) => {
                // 静态精确，但溢出了 → 用倍数
                (current_size as f64 * self.plan.growth_factor) as usize
            }
            DecisionSource::Pgo(_) => {
                // PGO 数据驱动
                ewma_suggested.max(
                    (current_size as f64 * self.plan.growth_factor) as usize
                )
            }
            _ => {
                // 自适应
                let min_required = needed + align;
                ewma_suggested.max(min_required).max(current_size * 2)
            }
        };
        
        // 检查最大限制
        if new_size > self.plan.max_size {
            return Err(AllocError::TooLarge {
                requested: new_size,
                max: self.plan.max_size,
            });
        }
        
        let new_block = MemoryBlock::new(new_size)?;
        
        // 记录扩容事件
        let event = GrowthEvent {
            timestamp: Instant::now(),
            old_size: current_size,
            new_size,
            reason: format!("needed {} bytes, had {}", needed, self.blocks[self.current_block].remaining()),
        };
        self.growth_history.push(event);
        self.stats.growth_count += 1;
        
        // 更新 EWMA
        self.ewma_state.update(new_size as f64);
        
        self.blocks.push(new_block);
        self.current_block = self.blocks.len() - 1;
        
        Ok(())
    }
    
    /// 计算下次扩容大小（供编译器静态分析调用）
    pub fn next_growth_size(&self, needed: usize) -> usize {
        let current = self.blocks[self.current_block].size;
        let predicted = self.ewma_state.current();
        
        let ewma_based = (predicted * 1.5) as usize;
        let factor_based = (current as f64 * self.plan.growth_factor) as usize;
        
        ewma_based.max(factor_based).max(needed * 2)
    }
    
    /// 当前已用内存
    pub fn current_used(&self) -> usize {
        self.blocks.iter().map(|b| b.used()).sum()
    }
    
    /// 当前总容量
    pub fn current_capacity(&self) -> usize {
        self.blocks.iter().map(|b| b.size).sum()
    }
    
    /// 碎片率（0.0 - 1.0）
    pub fn fragmentation_ratio(&self) -> f64 {
        let total_frag: usize = self.blocks.iter().map(|b| b.fragmentation).sum();
        let total_cap = self.current_capacity();
        
        if total_cap == 0 { 0.0 } else { total_frag as f64 / total_cap as f64 }
    }
    
    /// 销毁区域
    pub unsafe fn destroy(&mut self) {
        // 1. 逆序调用析构函数
        for entry in self.destructors.iter().rev() {
            (entry.drop_fn)(entry.ptr);
            self.stats.destructor_calls += 1;
        }
        self.destructors.clear();
        
        // 2. 释放所有内存块
        self.blocks.clear();  // Drop 会释放内存
        
        // 3. 清空 transferred
        self.transferred.clear();
        
        // 4. 更新统计
        self.stats.total_deallocated = self.stats.total_allocated;
    }
    
    /// 标记对象为已转移
    pub fn mark_transferred(&mut self, ptr: NonNull<u8>) {
        // 从析构列表中移除
        if let Some(pos) = self.destructors.iter().position(|e| e.ptr == ptr) {
            self.destructors.swap_remove(pos);
        }
        self.transferred.push(ptr);
    }
    
    /// 获取统计信息
    pub fn stats(&self) -> &RegionStats {
        &self.stats
    }
    
    /// 导出 PGO 数据
    pub fn export_pgo_data(&self) -> crate::profile::RegionProfile {
        let samples = &self.allocation_samples;
        let mut sorted = samples.clone();
        sorted.sort();
        
        let sum: usize = sorted.iter().sum();
        let avg = if sorted.is_empty() { 0.0 } else { sum as f64 / sorted.len() as f64 };
        
        let p50 = percentile(&sorted, 0.5);
        let p95 = percentile(&sorted, 0.95);
        let p99 = percentile(&sorted, 0.99);
        
        crate::profile::RegionProfile {
            call_count: self.stats.allocation_count as u64,
            size_stats: crate::profile::SizeStats {
                min: sorted.first().copied().unwrap_or(0),
                max: sorted.last().copied().unwrap_or(0),
                avg,
                p50,
                p95,
                p99,
            },
            growth_stats: crate::profile::GrowthStats {
                avg_growth_count: self.stats.growth_count as f64,
                avg_growth_factor: self.plan.growth_factor,
                max_growth_factor: self.plan.growth_factor,
            },
            samples: sorted,
        }
    }
    
    /// 决策来源（调试用）
    pub fn decision_source(&self) -> &DecisionSource {
        &self.plan.decision_source
    }
}

/// 计算百分位数
fn percentile(sorted: &[usize], p: f64) -> usize {
    if sorted.is_empty() {
        return 0;
    }
    let idx = (sorted.len() as f64 * p) as usize;
    let idx = idx.min(sorted.len() - 1);
    sorted[idx]
}

/// 检查类型是否需要析构
fn needs_drop<T>() -> bool {
    // 大多数类型需要析构（除了 Copy 类型）
    // 实际实现中应该更精确
    true
}

/// 安全的 drop_in_place
unsafe fn drop_in_place<T>(ptr: NonNull<u8>) {
    let typed_ptr = ptr.as_ptr() as *mut T;
    std::ptr::drop_in_place(typed_ptr);
}
```

```rust
// crates/zeta-region-alloc/src/compiler_interface.rs

use crate::static_sizer::{AllocSite, ExecFrequency, StaticSizer};

/// 编译器传递给分配器的信息
#[derive(Debug, Clone)]
pub struct RegionCompileInfo {
    /// 区域唯一标识

    pgo_data: Option<PgoData>,
    regions: HashMap<String, RegionCompileInfo>,

            pgo_data: None,
            regions: HashMap::new(),
        }

    ///

            .with_static(info.sizer.clone());

    }

                report.push_str(&format!("  Max: {} bytes\n", plan.max_size));
                report.push_str(&format!("  Growth Factor: {:.1}x\n", plan.growth_factor));
            }
            report.push('\n');
        }
        
        report
    }
```

```rust
#[test]
fn test_destructor_calls() {
    use std::sync::atomic::{AtomicUsize, Ordering};
    
    static DROP_COUNT: AtomicUsize = AtomicUsize::new(0);
    
    struct TrackDrop;
    impl Drop for TrackDrop {
        fn drop(&mut self) {
            DROP_COUNT.fetch_add(1, Ordering::SeqCst);
        }
    }
    
    let advisor = SizeAdvisor::new("r4".into());
    let mut region = SmartRegion::new("r4".into(), advisor).unwrap();
    
    for _ in 0..50 {
        region.allocate(TrackDrop).unwrap();
    }
    
    assert_eq!(DROP_COUNT.load(Ordering::SeqCst), 0);  // 还没销毁
    
    unsafe { region.destroy(); }
    
    assert_eq!(DROP_COUNT.load(Ordering::SeqCst), 50);  // 全部析构
}
```
