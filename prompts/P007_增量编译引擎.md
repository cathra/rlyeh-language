> **模块路径**：`crates/zeta-driver/`

## 任务描述

实现 Zeta 编译器的增量编译引擎：
1. **模块接口哈希**：检测模块接口是否变化
2. **编译产物缓存**：缓存 HIR/MIR/LIR 各级中间表示
3. **依赖图分析**：仅重新编译受影响的模块
4. **并行编译**：无依赖关系的模块同时编译

### 接口哈希计算

```rust
// crates/zeta-driver/src/incremental/hash.rs

use std::collections::BTreeMap;
use sha2::{Digest, Sha256};

/// 模块接口描述（用于哈希）
#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub struct ModuleInterface {
    /// 模块路径
    pub path: String,
    /// 公开项的签名哈希
    pub items: BTreeMap<String, ItemSignature>,
    /// 依赖模块的接口哈希
    pub dependencies: BTreeMap<String, String>, // name → hash
}

/// 项的签名（决定接口兼容性）
#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub enum ItemSignature {
    Function {
        name: String,
        params: Vec<TypeRepr>,
        ret: TypeRepr,
        generics: Vec<GenericParam>,
    },
    Struct {
        name: String,
        fields: BTreeMap<String, TypeRepr>,
        generics: Vec<GenericParam>,
    },
    Enum {
        name: String,
        variants: BTreeMap<String, Vec<TypeRepr>>,
    },
    Trait {
        name: String,
        methods: BTreeMap<String, MethodSignature>,
    },
    TypeAlias {
        name: String,
        target: TypeRepr,
    },
    Const {
        name: String,
        ty: TypeRepr,
    },
}

#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
pub enum TypeRepr {
    Named(String, Vec<TypeRepr>),
    Ref(Box<TypeRepr>, bool),  // mut?
    Array(Box<TypeRepr>, usize),
    Slice(Box<TypeRepr>),
    Tuple(Vec<TypeRepr>),
    Never,
}

/// 计算模块接口哈希
pub fn compute_interface_hash(iface: &ModuleInterface) -> String {
    let mut hasher = Sha256::new();
    // 序列化接口
    let bytes = bincode::serialize(iface).unwrap();
    hasher.update(&bytes);
    // 同时哈希依赖的接口哈希
    for (_, dep_hash) in &iface.dependencies {
        hasher.update(dep_hash.as_bytes());
    }
    format!("{:x}", hasher.finalize())
}

/// 检查模块是否需要重编译
pub fn needs_recompile(
    module_path: &str,
    cache_dir: &std::path::Path,
    current_iface: &ModuleInterface,
) -> bool {
    let hash_file = cache_dir.join(module_path).join("ast.hash");
    
    if !hash_file.exists() {
        return true;  // 没有缓存，需要编译
    }
    
    let cached: ModuleInterface = {
        let bytes = std::fs::read(&hash_file).unwrap();
        bincode::deserialize(&bytes).unwrap()
    };
    
    // 比较接口
    let current_hash = compute_interface_hash(current_iface);
    let cached_hash = compute_interface_hash(&cached);
    
    current_hash != cached_hash
}
```

### 依赖图与拓扑排序

```rust
// crates/zeta-driver/src/incremental/depgraph.rs

use std::collections::{HashMap, HashSet, VecDeque};

/// 模块依赖图
pub struct DependencyGraph {
    /// module → 它依赖的模块集合
    edges: HashMap<String, HashSet<String>>,
    /// module → 依赖它的模块集合（反向边）
    reverse_edges: HashMap<String, HashSet<String>>,
}

impl DependencyGraph {
    pub fn new() -> Self {
        Self {
            edges: HashMap::new(),
            reverse_edges: HashMap::new(),
        }
    }
    
    /// 添加依赖关系：from 依赖 to
    pub fn add_edge(&mut self, from: String, to: String) {
        self.edges.entry(from.clone()).or_default().insert(to.clone());
        self.reverse_edges.entry(to).or_default().insert(from);
    }
    
    /// 计算受影响模块集合
    /// 当某模块接口变化时，所有依赖它的模块都需要重编译
    pub fn affected_modules(&self, changed: &str) -> HashSet<String> {
        let mut affected = HashSet::new();
        let mut queue = VecDeque::new();
        queue.push_back(changed.to_string());
        
        while let Some(current) = queue.pop_front() {
            if let Some(deps) = self.reverse_edges.get(&current) {
                for dep in deps {
                    if affected.insert(dep.clone()) {
                        queue.push_back(dep.clone());
                    }
                }
            }
        }
        
        affected
    }
    
    /// 拓扑排序（用于编译顺序）
    pub fn topo_sort(&self) -> Vec<String> {
        let mut in_degree: HashMap<String, usize> = HashMap::new();
        
        for (from, tos) in &self.edges {
            in_degree.entry(from.clone()).or_insert(0);
            for to in tos {
                *in_degree.entry(to.clone()).or_insert(0) += 1;
            }
        }
        
        let mut queue: VecDeque<String> = in_degree
            .iter()
            .filter(|(_, &deg)| deg == 0)
            .map(|(name, _)| name.clone())
            .collect();
        
        let mut result = vec![];
        
        while let Some(node) = queue.pop_front() {
            result.push(node.clone());
            
            if let Some(tos) = self.edges.get(&node) {
                for to in tos {
                    let deg = in_degree.get_mut(to).unwrap();
                    *deg -= 1;
                    if *deg == 0 {
                        queue.push_back(to.clone());
                    }
                }
            }
        }
        
        result
    }
}
```

### 并行编译调度器

```rust
// crates/zeta-driver/src/incremental/scheduler.rs

use rayon::prelude::*;
use std::sync::Mutex;
use std::collections::HashSet;

/// 并行编译调度器
pub struct ParallelScheduler {
    /// 已完成编译的模块集合
    completed: Mutex<HashSet<String>>,
    /// 编译任务队列
    ready_queue: Mutex<std::collections::VecDeque<String>>,
}

impl ParallelScheduler {
    pub fn new() -> Self {
        Self {
            completed: Mutex::new(HashSet::new()),
            ready_queue: Mutex::new(std::collections::VecDeque::new()),
        }
    }
    
    /// 执行并行编译
    /// 
    /// # 算法
    /// 1. 计算拓扑排序
    /// 2. 将无依赖的模块放入就绪队列
    /// 3. 使用 rayon 并行处理就绪队列
    /// 4. 完成一个模块后，检查其依赖者是否就绪
    /// 5. 将新就绪的模块加入队列
    pub fn compile_all(
        &self,
        graph: &DependencyGraph,
        modules: &HashMap<String, ModuleSource>,
        cache: &IncrementalCache,
    ) -> Result<CompilationResult, CompileError> {
        let order = graph.topo_sort();
        let mut ready: Vec<String> = order
            .iter()
            .filter(|m| !modules[m].is_cached(cache))
            .filter(|m| {
                // 没有未编译的依赖
                graph.edges.get(*m).map_or(true, |deps| {
                    deps.iter().all(|d| {
                        modules.get(d).map_or(false, |src| src.is_cached(cache))
                    })
                })
            })
            .cloned()
            .collect();
        
        // 并行编译就绪模块
        let results: Result<Vec<_>, _> = ready
            .par_iter()
            .map(|module_name| {
                let source = &modules[module_name];
                compile_module(source, cache)
            })
            .collect();
        
        let compiled = results?;
        
        Ok(CompilationResult {
            compiled_modules: compiled,
            total_time: std::time::Duration::default(), // 实际应测量
        })
    }
}
```

### 缓存读写

```rust
// crates/zeta-driver/src/incremental/cache.rs

use std::path::Path;

/// 增量编译缓存
pub struct IncrementalCache {
    root: std::path::PathBuf,
}

impl IncrementalCache {
    pub fn new(project_root: &Path) -> Self {
        let root = project_root.join(".zeta_cache");
        std::fs::create_dir_all(&root).unwrap();
        Self { root }
    }
    
    /// 保存模块的 HIR 缓存
    pub fn save_hir(&self, module: &str, hir: &HirProgram) {
        let path = self.module_dir(module).join("hir.cache");
        let bytes = bincode::serialize(hir).unwrap();
        std::fs::write(path, bytes).unwrap();
    }
    
    /// 加载模块的 HIR 缓存
    pub fn load_hir(&self, module: &str) -> Option<HirProgram> {
        let path = self.module_dir(module).join("hir.cache");
        if !path.exists() {
            return None;
        }
        let bytes = std::fs::read(path).ok()?;
        bincode::deserialize(&bytes).ok()
    }
    
    /// 保存接口哈希
    pub fn save_interface(&self, module: &str, iface: &ModuleInterface) {
        let path = self.module_dir(module).join("ast.hash");
        let bytes = bincode::serialize(iface).unwrap();
        std::fs::write(path, bytes).unwrap();
    }
    
    /// 加载接口哈希
    pub fn load_interface(&self, module: &str) -> Option<ModuleInterface> {
        let path = self.module_dir(module).join("ast.hash");
        let bytes = std::fs::read(path).ok()?;
        bincode::deserialize(&bytes).ok()
    }
    
    /// 保存 PGO 数据
    pub fn save_pgo_data(&self, data: &PgoData) {
        let path = self.root.join("profile.json");
        let json = serde_json::to_string_pretty(data).unwrap();
        std::fs::write(path, json).unwrap();
    }
    
    /// 加载 PGO 数据
    pub fn load_pgo_data(&self) -> Option<PgoData> {
        let path = self.root.join("profile.json");
        let json = std::fs::read_to_string(path).ok()?;
        serde_json::from_str(&json).ok()
    }
    
    fn module_dir(&self, module: &str) -> std::path::PathBuf {
        let dir = self.root.join("modules").join(module.replace("::", "_"));
        std::fs::create_dir_all(&dir).unwrap();
        dir
    }
    
    /// 清理过期缓存
    pub fn clean_stale(&self, active_modules: &HashSet<String>) {
        let modules_dir = self.root.join("modules");
        if !modules_dir.exists() {
            return;
        }
        
        for entry in std::fs::read_dir(&modules_dir).unwrap() {
            let entry = entry.unwrap();
            let name = entry.file_name().to_string_lossy().to_string();
            if !active_modules.contains(&name) {
                std::fs::remove_dir_all(entry.path()).unwrap();
            }
        }
    }
}
```

### 编译器驱动

```rust
// crates/zeta-driver/src/lib.rs

#![warn(missing_docs)]

use std::path::Path;
use std::time::Instant;

/// 编译器驱动
pub struct ZetaDriver {
    cache: IncrementalCache,
    scheduler: ParallelScheduler,
    verbose: bool,
}

/// 编译结果
pub struct CompilationResult {
    pub compiled_modules: Vec<String>,
    pub total_time: std::time::Duration,
    pub cache_hits: usize,
    pub cache_misses: usize,
}

impl ZetaDriver {
    pub fn new(project_root: &Path) -> Self {
        Self {
            cache: IncrementalCache::new(project_root),
            scheduler: ParallelScheduler::new(),
            verbose: false,
        }
    }
    
    pub fn with_verbose(mut self, v: bool) -> Self {
        self.verbose = v;
        self
    }
    
    /// 执行完整编译流程
    pub fn compile(&self, project: &Project) -> Result<CompilationResult, CompileError> {
        let start = Instant::now();
        
        // 1. 解析所有源文件
        let mut modules = HashMap::new();
        for source_file in &project.sources {
            let source = std::fs::read_to_string(&source_file)?;
            let mut parser = Parser::new(&source)?;
            let ast = parser.parse_program()?;
            
            let module_name = source_file.stem().unwrap().to_string_lossy().to_string();
            modules.insert(module_name, ModuleSource { ast, source });
        }
        
        // 2. 计算接口哈希，确定哪些模块需要重编译
        let mut to_compile = vec![];
        let mut cache_hits = 0;
        let mut cache_misses = 0;
        
        for (name, module) in &modules {
            let iface = extract_interface(&module.ast);
            let cached = self.cache.load_interface(name);
            
            let needs_recompile = match cached {
                Some(cached_iface) => {
                    compute_interface_hash(&iface) != compute_interface_hash(&cached_iface)
                }
                None => true,
            };
            
            if needs_recompile {
                to_compile.push(name.clone());
                cache_misses += 1;
            } else {
                cache_hits += 1;
            }
        }
        
        if self.verbose {
            println!("[Zeta] Cache hits: {}, misses: {}", cache_hits, cache_misses);
        }
        
        // 3. 构建依赖图
        let mut graph = DependencyGraph::new();
        for (name, module) in &modules {
            for dep in extract_dependencies(&module.ast) {
                graph.add_edge(name.clone(), dep);
            }
        }
        
        // 4. 确定受影响模块
        let mut affected = HashSet::new();
        for changed in &to_compile {
            affected.extend(graph.affected_modules(changed));
        }
        
        // 5. 并行编译
        let affected_set: HashSet<_> = affected.iter().cloned().collect();
        let modules_to_compile: Vec<_> = to_compile
            .into_iter()
            .chain(affected_set.into_iter())
            .collect();
        
        let compiled = self.compile_modules(&modules_to_compile, &modules, &graph)?;
        
        // 6. 代码生成 + 链接
        self.link(&compiled)?;
        
        let total_time = start.elapsed();
        
        if self.verbose {
            println!("[Zeta] Compiled {} modules in {:?}", compiled.len(), total_time);
        }
        
        Ok(CompilationResult {
            compiled_modules: compiled.iter().map(|m| m.name.clone()).collect(),
            total_time,
            cache_hits,
            cache_misses,
        })
    }
    
    fn compile_modules(
        &self,
        names: &[String],
        modules: &HashMap<String, ModuleSource>,
        graph: &DependencyGraph,
    ) -> Result<Vec<CompiledModule>, CompileError> {
        // 按拓扑顺序编译
        let order = graph.topo_sort();
        let ordered: Vec<_> = order
            .into_iter()
            .filter(|n| names.contains(n))
            .collect();
        
        let mut compiled = vec![];
        
        for name in ordered {
            let module = &modules[&name];
            
            // 类型检查
            let hir = typecheck(module.ast.clone())?;
            
            // 保存缓存
            self.cache.save_hir(&name, &hir);
            
            // 区域检查
            let checked = region_check(hir)?;
            
            // MIR lowering
            let mir = lower_to_mir(checked)?;
            
            // 代码生成
            let codegen = generate_code(mir)?;
            
            compiled.push(CompiledModule {
                name: name.clone(),
                object_code: codegen,
            });
        }
        
        Ok(compiled)
    }
    
    fn link(&self, modules: &[CompiledModule]) -> Result<(), CompileError> {
        // 调用链接器（如 ld 或 lld）
        todo!("链接所有目标文件")
    }
}

    ast: AstProgram,
    source: String,
}

    object_code: Vec<u8>,
}
```

```rust
    // 相同接口应该产生相同哈希
```
