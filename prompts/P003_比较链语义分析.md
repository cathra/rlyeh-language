## 任务描述

## 核心语义规则

### 比较链方向规则

```
正向链（区间内）：所有运算符为 < 或 <=
    0 < x < 10    →  x > 0 && x < 10
    0 <= x <= 10  →  x >= 0 && x <= 10
    方向一致，OK

反向链（区间外）：所有运算符为 > 或 >=
    0 > x > 10    →  x < 0 || x > 10
    0 >= x >= 10  →  x <= 0 || x >= 10
    方向一致，OK

混合方向 → 编译错误
    0 < x > 10    →  ERROR: inconsistent comparison direction
```

### `in` 表达式展开规则

```
x in (1, 3, 5)           →  x == 1 || x == 3 || x == 5
x in (0..10)              →  x >= 0 && x < 10
x in [0, 10]              →  x >= 0 && x <= 10
x in (0, 10]              →  x > 0 && x <= 10
x not in (1, 2, 3)        →  x != 1 && x != 2 && x != 3
x in (1..10, 20, 30)      →  (x >= 1 && x < 10) || x == 20 || x == 30
```

```rust
// crates/zeta-typecheck/src/lib.rs

//! Zeta 类型检查器
//!
//! 对 AST 进行类型检查和语义验证，输出类型标注的 HIR。

#![warn(missing_docs)]
#![warn(unsafe_code)]

use zeta_ast::*;
use zeta_hir::*;
use thiserror::Error;
use std::collections::HashMap;

/// 类型环境
pub struct TypeContext {
    /// 变量 → 类型映射
    pub variables: HashMap<String, Type>,
    /// 当前函数返回类型
    pub return_type: Option<Type>,
    /// 已知类型定义
    pub type_defs: HashMap<String, TypeDef>,
    /// 错误列表（支持多错误报告）
    pub errors: Vec<TypeError>,
}

/// 类型定义
pub enum TypeDef {
    Struct(StructDef),
    Enum(EnumDef),
    Trait(TraitDef),
    TypeAlias(String),  // 类型别名名
}

/// 类型
#[derive(Debug, Clone, PartialEq)]
pub enum Type {
    /// 基本类型
    I8, I16, I32, I64, I128, ISize,
    U8, U16, U32, U64, U128, USize,
    F32, F64,
    Bool, Char, Str,
    
    /// 指针/引用
    Ref(Box<Type>, Mutability),
    RawPtr(Box<Type>, Mutability),
    
    /// 复合类型
    Array(Box<Type>, usize),
    Slice(Box<Type>),
    Tuple(Vec<Type>),
    
    /// 用户定义类型
    Named(String, Vec<Type>),  // 名称 + 泛型参数
    
    /// 特殊
    Never,   // !
    Infer,    // 待推断
    Generic(String),  // 泛型参数 T
}

/// 可变性
#[derive(Debug, Clone, PartialEq)]
pub enum Mutability {
    Mutable,
    Immutable,
}

/// 类型错误
#[derive(Debug, Error)]
pub enum TypeError {
    #[error("type mismatch: expected {expected}, found {found} at {line}:{col}")]
    TypeMismatch {
        expected: String,
        found: String,
        line: usize,
        col: usize,
    },
    
    #[error("cannot find value '{name}' at {line}:{col}")]
    UndefinedValue { name: String, line: usize, col: usize },
    
    #[error("inconsistent comparison direction at {line}:{col}: {detail}")]
    InconsistentComparison {
        detail: String,
        line: usize,
        col: usize,
    },
    
    #[error("comparison chain requires PartialOrd at {line}:{col}")]
    MissingPartialOrd {
        line: usize,
        col: usize,
    },
    
    #[error("incompatible types in comparison chain: {left_ty} and {right_ty} at {line}:{col}")]
    ChainTypeMismatch {
        left_ty: String,
        right_ty: String,
        line: usize,
        col: usize,
    },
    
    #[error("invalid 'in' set: element types must match value type at {line}:{col}")]
    InSetTypeMismatch {
        line: usize,
        col: usize,
    },
    
    #[error("operator '{op}' cannot be applied to type {ty} at {line}:{col}")]
    InvalidOpForType {
        op: String,
        ty: String,
        line: usize,
        col: usize,
    },
}

impl TypeContext {
    pub fn new() -> Self {
        let mut ctx = Self {
            variables: HashMap::new(),
            return_type: None,
            type_defs: HashMap::new(),
            errors: vec![],
        };
        ctx.insert_builtin_types();
        ctx
    }
    
    fn insert_builtin_types(&mut self) {
        // 内置类型定义
    }
}

/// 类型检查入口
pub fn typecheck(program: AstProgram) -> Result<HirProgram, Vec<TypeError>> {
    let mut ctx = TypeContext::new();
    let mut hir_items = vec![];
    
    for item in program.items {
        match ctx.check_item(item) {
            Ok(hir_item) => hir_items.push(hir_item),
            Err(e) => ctx.errors.push(e),
        }
    }
    
    if ctx.errors.is_empty() {
        Ok(HirProgram { items: hir_items })
    } else {
        Err(ctx.errors)
    }
}
```

```rust
impl TypeContext {
    /// 推断表达式类型
    fn infer_expr(&mut self, expr: AstExpr) -> Result<(HirExpr, Type), TypeError> {
        match expr.kind {
            ExprKind::IntLiteral(n) => {
                // 根据值大小推断具体整数类型
                let ty = if n >= i64::MIN as i128 && n <= i64::MAX as i128 {
                    Type::I64
                } else {
                    Type::I128
                };
                Ok((HirExpr::int_literal(n, ty.clone()), ty))
            }
            
            ExprKind::FloatLiteral(f) => {
                Ok((HirExpr::float_literal(f, Type::F64), Type::F64))
            }
            
            ExprKind::StringLiteral(s) => {
                Ok((HirExpr::string_literal(s), Type::Str))
            }
            
            ExprKind::BoolLiteral(b) => {
                Ok((HirExpr::bool_literal(b), Type::Bool))
            }
            
            ExprKind::Ident(name) => {
                let ty = self.variables.get(&name)
                    .ok_or(TypeError::UndefinedValue { name, line: 0, col: 0 })?
                    .clone();
                Ok((HirExpr::variable(name, ty.clone()), ty))
            }
            
            ExprKind::ComparisonChain { elements, operators } => {
                self.check_comparison_chain(elements, operators, span)
                    .map(|hir| (hir, Type::Bool))
            }
            
            ExprKind::InSet { value, set } => {
                self.check_in_expression(*value, set, false, span)
                    .map(|hir| (hir, Type::Bool))
            }
            
            // ... 其他表达式类型
            _ => todo!("其他表达式的类型推断"),
        }
    }
}
```

## 测试用例

```rust
// crates/zeta-typecheck/tests/typecheck_test.rs

use zeta_typecheck::{typecheck, TypeError};
use zeta_parser::Parser;

fn check(source: &str) -> Result<(), Vec<TypeError>> {
    let mut parser = Parser::new(source).unwrap();
    let ast = parser.parse_program().unwrap();
    typecheck(ast).map(|_| ())
}

#[test]
fn test_forward_chain() {
    // 0 < x < 10 → 应该通过
    let result = check("fn main() { let x = 5; if 0 < x < 10 {} }");
    assert!(result.is_ok());
}

#[test]
fn test_backward_chain() {
    // 0 > x > 10 → 应该通过（x < 0 || x > 10）
    let result = check("fn main() { let x = -5; if 0 > x > 10 {} }");
    assert!(result.is_ok());
}

#[test]
fn test_inconsistent_chain() {
    // 0 < x > 10 → 应该报错
    let errs = check("fn main() { let x = 5; if 0 < x > 10 {} }");
    assert!(errs.is_err());
    let errs = errs.unwrap_err();
    assert!(errs.iter().any(|e| matches!(e, TypeError::InconsistentComparison { .. })));
}

#[test]
fn test_in_discrete_set() {
    let result = check(r#"fn main() { let x = 3; if x in (1, 3, 5) {} }"#);
    assert!(result.is_ok());
}

#[test]
fn test_in_range() {
    let result = check("fn main() { let x = 5; if x in (0..10) {} }");
    assert!(result.is_ok());
}

#[test]
fn test_not_in() {
    let result = check("fn main() { let x = 5; if x not in (1, 2, 3) {} }");
    assert!(result.is_ok());
}

#[test]
fn test_in_type_mismatch() {
    // x (i32) in ("a", "b") → 类型不匹配
    let errs = check(r#"fn main() { let x = 5; if x in ("a", "b") {} }"#);
    assert!(errs.is_err());
}

#[test]
fn test_between_default() {
    // x between 0 and 10 → 0 <= x <= 10
    let result = check("fn main() { let x = 5; if x between 0 and 10 {} }");
    assert!(result.is_ok());
}

#[test]
fn test_between_exclusive() {
    let result = check("fn main() { let x = 5; if x between exclusive 0 and 10 {} }");
    assert!(result.is_ok());
}

#[test]
fn test_between_with_time() {
    let result = check("fn main() { let h = 14; if h between 9am and 6pm {} }");
    assert!(result.is_ok());
}

#[test]
fn test_chain_with_different_types() {
    // 0 < x < 10.5 → f64 和 i32 混合，应该通过（隐式转换）

    assert!(result.is_ok());

#[test]
fn test_nested_comparison() {
    let result = check("fn main() { let x = 5; if (0 < x < 10) && (x != 7) {} }");
```

```rust
// crates/zeta-typecheck/benches/typecheck_bench.rs
use criterion::{black_box, Criterion};

fn bench_check_simple(c: &mut Criterion) {
    let source = "fn main() { let x = 5; if 0 < x < 10 { println!(\"{}\", x); } }";
    c.bench_function("check_comparison_chain", |b| {
        b.iter(|| {
            let mut parser = Parser::new(black_box(source)).unwrap();
            let ast = parser.parse_program().unwrap();
            typecheck(ast).unwrap()
        })
    });
}
```
