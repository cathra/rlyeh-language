## 核心类型

### Option 和 Result

```zeta
/// 可选值

    Some(T),
    None,
}

impl<T> Option<T> {
    pub fn is_some(&self) -> bool { ... }
    pub fn is_none(&self) -> bool { ... }
    pub fn unwrap(self) -> T { ... }
    pub fn map<U>(self, f: Fn(T) -> U) -> Option<U> { ... }
    pub fn and_then<U>(self, f: Fn(T) -> Option<U>) -> Option<U> { ... }
    pub fn or_else(self, f: Fn() -> Option<T>) -> Option<T> { ... }
}

/// 结果类型
enum Result<T, E> {
    Ok(T),
    Err(E),
}

impl<T, E> Result<T, E> {
    pub fn is_ok(&self) -> bool { ... }
    pub fn is_err(&self) -> bool { ... }
    pub fn unwrap(self) -> T { ... }
    pub fn expect(self, msg: &str) -> T { ... }
    pub fn map<U>(self, f: Fn(T) -> U) -> Result<U, E> { ... }
    pub fn map_err<F>(self, f: Fn(E) -> F) -> Result<T, F> { ... }
}
```

## 集合类型

### Vec

```zeta
pub struct Vec<T, A: Allocator = Global> {
    ptr: NonNull<T>,
    len: usize,
    cap: usize,
    alloc: A,
}

impl<T> Vec<T> {
    pub fn new() -> Self { ... }
    pub fn with_capacity(cap: usize) -> Self { ... }
    
    pub fn push(&mut self, value: T) {
        if self.len == self.cap {
            self.grow();
        }
        unsafe {
            self.ptr.as_ptr().add(self.len).write(value);
        }
        self.len += 1;
    }
    
    pub fn pop(&mut self) -> Option<T> { ... }
    pub fn get(&self, index: usize) -> Option<&T> { ... }
    pub fn iter(&self) -> Iter<T> { ... }
}
```

pub async fn read_file(path: &str) -> Result<String, IoError> {
    let mut file = File::open(path)?;
    let mut buf = String::new();
    file.read_to_string(&mut buf).await?;
    Ok(buf)
}

## 序列化模块

### JSON

```zeta
pub trait Serialize {
    fn serialize(&self) -> JsonValue;
}

pub trait Deserialize: Sized {
    fn deserialize(value: &JsonValue) -> Result<Self, JsonError>;
}

// 自动派生
#[derive(Serialize, Deserialize)]
pub struct User {
    pub id: u32,
    pub name: String,
    pub email: String,
}

// 使用
let json = user.serialize().to_string();
let user2: User = User::deserialize(&json.parse()?)?;
```
