/// 可选值：有值(Some)或无值(None)
pub enum Option<T> {
    Some(T),
    None,
}

impl<T> Option<T> {
    /// 创建 Some(value)
    pub fn some(value: T) -> Option<T> {
        Option::Some(value)
    }
    
    /// 创建 None
    pub fn none() -> Option<T> {
        Option::None
    }
    
    /// 解包，若为 None 则 panic
    pub fn unwrap(self) -> T {
        match self {
            Option::Some(v) => v,
            Option::None => panic("called unwrap on None"),
        }
    }
    
    /// 解包，若为 None 则返回默认值
    pub fn unwrap_or(self, default: T) -> T {
        match self {
            Option::Some(v) => v,
            Option::None => default,
        }
    }
    
    /// 映射内部值
    pub fn map<U>(self, f: fn(T) -> U) -> Option<U> {
        match self {
            Option::Some(v) => Option::Some(f(v)),
            Option::None => Option::None,
        }
    }
    
    /// 扁平映射
    pub fn and_then<U>(self, f: fn(T) -> Option<U>) -> Option<U> {
        match self {
            Option::Some(v) => f(v),
            Option::None => Option::None,
        }
    }
    
    /// 是否为 Some
    pub fn is_some(&self) -> bool {
        match self {
            Option::Some(_) => true,
            Option::None => false,
        }
    }
    
    /// 是否为 None
    pub fn is_none(&self) -> bool {
        match self {
            Option::Some(_) => false,
            Option::None => true,
        }
    }
    
    /// 模式匹配辅助
    pub fn match<R>(self, some_fn: fn(T) -> R, none_fn: fn() -> R) -> R {
        match self {
            Option::Some(v) => some_fn(v),
            Option::None => none_fn(),
        }
    }
}

/// 结果类型：成功(Ok)或错误(Err)
pub enum Result<T, E> {
    Ok(T),
    Err(E),
}

impl<T, E> Result<T, E> {
    /// 解包 Ok 值，若为 Err 则 panic
    pub fn unwrap(self) -> T {
        match self {
            Result::Ok(v) => v,
            Result::Err(e) => panic_fmt("unwrap failed: {:?}", e),
        }
    }
    
    /// 解包 Ok 值，若为 Err 则返回默认值
    pub fn unwrap_or(self, default: T) -> T {
        match self {
            Result::Ok(v) => v,
            Result::Err(_) => default,
        }
    }
    
    /// 映射 Ok 值
    pub fn map<U>(self, f: fn(T) -> U) -> Result<U, E> {
        match self {
            Result::Ok(v) => Result::Ok(f(v)),
            Result::Err(e) => Result::Err(e),
        }
    }
    
    /// 映射 Err 值
    pub fn map_err<F>(self, f: fn(E) -> F) -> Result<T, F> {
        match self {
            Result::Ok(v) => Result::Ok(v),
            Result::Err(e) => Result::Err(f(e)),
        }
    }
    
    /// 扁平映射
    pub fn and_then<U>(self, f: fn(T) -> Result<U, E>) -> Result<U, E> {
        match self {
            Result::Ok(v) => f(v),
            Result::Err(e) => Result::Err(e),
        }
    }
    
    /// 错误传播（等价于 ? 运算符）
    pub fn propagate(self) -> Result<T, E> {
        self
    }
    
    /// 是否为 Ok
    pub fn is_ok(&self) -> bool {
        match self {
            Result::Ok(_) => true,
            Result::Err(_) => false,
        }
    }
    
    /// 是否为 Err
    pub fn is_err(&self) -> bool {
        match self {
            Result::Ok(_) => false,
            Result::Err(_) => true,
        }
    }
}

/// 迭代器 trait
pub trait Iterator {
    type Item;
    
    fn next(&mut self) -> Option<Self::Item>;
    
    /// 收集为集合
    fn collect<B: FromIterator<Self::Item>>(self) -> B
    where Self: Sized {
        B::from_iterator(self)
    }
    
    /// 映射
    fn map<B>(self, f: fn(Self::Item) -> B) -> Map<Self, B>
    where Self: Sized {
        Map { iter: self, f }
    }
    
    /// 过滤
    fn filter(self, pred: fn(&Self::Item) -> bool) -> Filter<Self>
    where Self: Sized {
        Filter { iter: self, pred }
    }
    
    /// 计数
    fn count(self) -> usize
    where Self: Sized {
        let mut c = 0;
        let mut iter = self;
        while iter.next().is_some() {
            c += 1;
        }
        c
    }
    
    /// 求和
    fn sum(self) -> Self::Item
    where Self: Sized, Self::Item: Add + Zero {
        let mut total = Self::Item::zero();
        let mut iter = self;
        while let Some(item) = iter.next() {
            total = total + item;
        }
        total
    }
    
    /// 查找第一个满足条件的元素
    fn find(self, pred: fn(&Self::Item) -> bool) -> Option<Self::Item>
    where Self: Sized {
        let mut iter = self;
        while let Some(item) = iter.next() {
            if pred(&item) {
                return Option::Some(item);
            }
        }
        Option::None
    }
    
    /// 是否所有元素满足条件
    fn all(self, pred: fn(&Self::Item) -> bool) -> bool
    where Self: Sized {
        let mut iter = self;
        while let Some(item) = iter.next() {
            if !pred(&item) {
                return false;
            }
        }
        true
    }
    
    /// 是否任一元素满足条件
    fn any(self, pred: fn(&Self::Item) -> bool) -> bool
    where Self: Sized {
        let mut iter = self;
        while let Some(item) = iter.next() {
            if pred(&item) {
                return true;
            }
        }
        false
    }
    
    /// 取前 n 个元素
    fn take(self, n: usize) -> Take<Self>
    where Self: Sized {
        Take { iter: self, remaining: n }
    }
    
    /// 跳过前 n 个元素
    fn skip(self, n: usize) -> Skip<Self>
    where Self: Sized {
        Skip { iter: self, to_skip: n }
    }
    
    /// 折叠
    fn fold<B>(self, init: B, f: fn(B, Self::Item) -> B) -> B
    where Self: Sized {
        let mut acc = init;
        let mut iter = self;
        while let Some(item) = iter.next() {
            acc = f(acc, item);
        }
        acc
    }
}

/// 从迭代器构建集合
pub trait FromIterator<T> {
    fn from_iterator<I: Iterator<Item = T>>(iter: I) -> Self;
}

/// 动态数组
pub struct Vec<T> {
    ptr: *mut T,
    len: usize,
    cap: usize,
}

impl<T> Vec<T> {
    /// 创建空 Vec
    pub fn new() -> Vec<T> {
        Vec {
            ptr: std::mem::alloc(16 * std::mem::size_of::<T>() as usize) as *mut T,
            len: 0,
            cap: 16,
        }
    }
    
    /// 创建并预分配
    pub fn with_capacity(cap: usize) -> Vec<T> {
        Vec {
            ptr: std::mem::alloc(cap * std::mem::size_of::<T>() as usize) as *mut T,
            len: 0,
            cap,
        }
    }
    
    /// 在区域中创建
    pub fn new_in(region: &Region) -> Vec<T> {
        // 在指定区域分配底层存储
        todo!("区域分配集成")
    }
    
    /// 追加元素
    pub fn push(&mut self, value: T) {
        if self.len >= self.cap {
            self.grow();
        }
        unsafe {
            *self.ptr.add(self.len) = value;
        }
        self.len += 1;
    }
    
    /// 弹出元素
    pub fn pop(&mut self) -> Option<T> {
        if self.len == 0 {
            return Option::None;
        }
        self.len -= 1;
        unsafe {
            Option::Some(*self.ptr.add(self.len))
        }
    }
    
    /// 按索引访问
    pub fn get(&self, index: usize) -> Option<&T> {
        if index >= self.len {
            Option::None
        } else {
            unsafe { Option::Some(&*self.ptr.add(index)) }
        }
    }
    
    /// 按索引访问（可变）
    pub fn get_mut(&mut self, index: usize) -> Option<&mut T> {
        if index >= self.len {
            Option::None
        } else {
            unsafe { Option::Some(&mut *self.ptr.add(index)) }
        }
    }
    
    /// 长度
    pub fn len(&self) -> usize {
        self.len
    }
    
    /// 是否为空
    pub fn is_empty(&self) -> bool {
        self.len == 0
    }
    
    /// 容量
    pub fn capacity(&self) -> usize {
        self.cap
    }
    
    /// 清空
    pub fn clear(&mut self) {
        // 调用每个元素的析构
        for i in 0..self.len {
            unsafe {
                std::mem::drop_in_place(self.ptr.add(i));
            }
        }
        self.len = 0;
    }
    
    /// 排序
    pub fn sort(&mut self)
    where T: Ord {
        // 使用快速排序或归并排序
        self.quick_sort(0, self.len);
    }
    
    /// 二分查找
    pub fn binary_search(&self, value: &T) -> Result<usize, usize>
    where T: Ord {
        let mut low = 0;
        let mut high = self.len;
        
        while low < high {
            let mid = low + (high - low) / 2;
            let mid_val = unsafe { &*self.ptr.add(mid) };
            
            if mid_val < value {
                low = mid + 1;
            } else if mid_val > value {
                high = mid;
            } else {
                return Result::Ok(mid);
            }
        }
        
        Result::Err(low)
    }
    
    /// 迭代器
    pub fn iter(&self) -> Iter<T> {
        Iter {
            vec: self,
            index: 0,
        }
    }
    
    pub fn iter_mut(&mut self) -> IterMut<T> {
        IterMut {
            vec: self,
            index: 0,
        }
    }
    
    // ===== 内部方法 =====
    
    fn grow(&mut self) {
        let new_cap = self.cap * 2;
        let new_layout = std::alloc::Layout::array::<T>(new_cap).unwrap();
        let new_ptr = unsafe { std::alloc::alloc(new_layout) as *mut T };
        
        // 移动元素
        for i in 0..self.len {
            unsafe {
                *new_ptr.add(i) = *self.ptr.add(i);
            }
        }
        
        unsafe {
            let old_layout = std::alloc::Layout::array::<T>(self.cap).unwrap();
            std::alloc::dealloc(self.ptr as *mut u8, old_layout);
        }
        
        self.ptr = new_ptr;
        self.cap = new_cap;
    }
    
    fn quick_sort(&mut self, low: usize, high: usize)
    where T: Ord {
        if low >= high {
            return;
        }
        
        let pivot = unsafe { (*self.ptr.add(high - 1)).clone() };
        let mut i = low;
        
        for j in low..high - 1 {
            let j_val = unsafe { (*self.ptr.add(j)).clone() };
            if j_val <= pivot {
                unsafe {
                    std::mem::swap(self.ptr.add(i), self.ptr.add(j));
                }
                i += 1;
            }
        }
        
        unsafe {
            std::mem::swap(self.ptr.add(i), self.ptr.add(high - 1));
        }
        
        self.quick_sort(low, i);
        self.quick_sort(i + 1, high);
    }
}

/// Vec 的迭代器
pub struct Iter<'a, T> {
    vec: &'a Vec<T>,
    index: usize,
}

impl<'a, T> Iterator for Iter<'a, T> {
    type Item = &'a T;
    
    fn next(&mut self) -> Option<&'a T> {
        if self.index >= self.vec.len {
            Option::None
        } else {
            let result = unsafe { &*self.vec.ptr.add(self.index) };
            self.index += 1;
            Option::Some(result)
        }
    }
}

/// Vec 的可变迭代器
pub struct IterMut<'a, T> {
    vec: &'a mut Vec<T>,
    index: usize,
}

impl<'a, T> Iterator for IterMut<'a, T> {
    type Item = &'a mut T;
    
    fn next(&mut self) -> Option<&'a mut T> {
        if self.index >= self.vec.len {
            Option::None
        } else {
            let result = unsafe { &mut *self.vec.ptr.add(self.index) };
            self.index += 1;
            Option::Some(result)
        }
    }
}

/// Vec 的析构
impl<T> Drop for Vec<T> {
    fn drop(&mut self) {
        // 析构所有元素
        for i in 0..self.len {
            unsafe {
                std::mem::drop_in_place(self.ptr.add(i));
            }
        }
        // 释放内存
        unsafe {
            let layout = std::alloc::Layout::array::<T>(self.cap).unwrap();
            std::alloc::dealloc(self.ptr as *mut u8, layout);
        }
    }
}

    vec: Vec<u8>,

impl String {
    pub fn new() -> String {
        String { vec: Vec::new() }
    }
    
    pub fn from(s: &str) -> String {
        let mut result = String::new();
        for byte in s.bytes() {
            result.vec.push(byte);
        }
        result
    }
    
    /// 追加字符
    pub fn push(&mut self, ch: char) {
        let mut buf = [0u8; 4];
        let encoded = ch.encode_utf8(&mut buf);
        for byte in encoded.bytes() {
            self.vec.push(byte);
        }
    }
    
    /// 追加字符串
    pub fn push_str(&mut self, s: &str) {
        for byte in s.bytes() {
            self.vec.push(byte);
        }
    }
    
    /// 长度（字节）
    pub fn len(&self) -> usize {
        self.vec.len()
    }
    
    /// 是否为空
    pub fn is_empty(&self) -> bool {
        self.vec.is_empty()
    }
    
    /// 转为 &str
    pub fn as_str(&self) -> &str {
        unsafe {
            std::str::from_utf8_unchecked(&self.vec[..])
        }
    }
    
    /// 转为大写
    pub fn to_uppercase(&self) -> String {
        let mut result = String::new();
        for ch in self.chars() {
            for upper in ch.to_uppercase() {
                result.push(upper);
            }
        }
        result
    }
    
    /// 转为小写
    pub fn to_lowercase(&self) -> String {
        let mut result = String::new();
        for ch in self.chars() {
            for lower in ch.to_lowercase() {
                result.push(lower);
            }
        }
        result
    }
    
    /// 去除空白
    pub fn trim(&self) -> &str {
        let s = self.as_str();
        s.trim()
    }
    
    /// 字符迭代器
    pub fn chars(&self) -> Chars {
        Chars {
            string: self,
            position: 0,
        }
    }
    
    /// 按分隔符分割
    pub fn split(&self, sep: &str) -> Split {
        Split {
            string: self.as_str(),
            separator: sep,
            position: 0,
        }
    }
    
    /// 替换
    pub fn replace(&self, from: &str, to: &str) -> String {
        let mut result = String::new();
        let s = self.as_str();
        
        let mut start = 0;
        while let Some(pos) = s[start..].find(from) {
            result.push_str(&s[start..start + pos]);
            result.push_str(to);
            start += pos + from.len();
        }
        result.push_str(&s[start..]);
        
        result
    }
}

/// 文件句柄
pub struct File {
    fd: i32,
    path: String,
    mode: OpenMode,
}

pub enum OpenMode {
    Read,
    Write,
    Append,
    ReadWrite,
}

/// IO 错误
pub enum IoError {
    NotFound(String),
    PermissionDenied(String),
    ConnectionRefused,
    ConnectionReset,
    TimedOut,
    AlreadyExists(String),
    InvalidInput(String),
    Other(String),
}

impl File {
    /// 打开文件（只读）
    pub fn open(path: &str) -> Result<File, IoError> {
        let fd = unsafe { sys_open(path, 0 /* O_RDONLY */) };
        if fd < 0 {
            return Result::Err(IoError::NotFound(path.to_string()));
        }
        Result::Ok(File {
            fd,
            path: String::from(path),
            mode: OpenMode::Read,
        })
    }
    
    /// 创建文件（写入，截断）
    pub fn create(path: &str) -> Result<File, IoError> {
        let fd = unsafe { sys_create(path, 0o644) };
        if fd < 0 {
            return Result::Err(IoError::PermissionDenied(path.to_string()));
        }
        Result::Ok(File {
            fd,
            path: String::from(path),
            mode: OpenMode::Write,
        })
    }
    
    /// 打开（自定义模式）
    pub fn open_with(path: &str, mode: OpenMode) -> Result<File, IoError> {
        let flags = match mode {
            OpenMode::Read => 0,
            OpenMode::Write => 1 | 512,  // O_WRONLY | O_TRUNC
            OpenMode::Append => 1 | 1024, // O_WRONLY | O_APPEND
            OpenMode::ReadWrite => 2,       // O_RDWR
        };
        
        let fd = unsafe { sys_open(path, flags) };
        if fd < 0 {
            return Result::Err(IoError::NotFound(path.to_string()));
        }
        Result::Ok(File {
            fd,
            path: String::from(path),
            mode,
        })
    }
    
    /// 读取所有内容到 String
    pub fn read_to_string(&mut self) -> Result<String, IoError> {
        let mut result = String::new();
        let mut buffer = [0u8; 4096];
        
        loop {
            let bytes_read = unsafe {
                sys_read(self.fd, &mut buffer[0], buffer.len())
            };
            
            if bytes_read == 0 {
                break;  // EOF
            }
            if bytes_read < 0 {
                return Result::Err(IoError::Other("read failed".to_string()));
            }
            
            // 按 UTF-8 解码整块字节，而不是逐字节 as char（否则多字节字符被破坏）
            let bytes = &buffer[..bytes_read as usize];
            match std::str::from_utf8(bytes) {
                Ok(s) => result.push_str(s),
                Err(_) => {
                    return Result::Err(IoError::Other(
                        "invalid UTF-8 sequence in file".to_string(),
                    ))
                }
            }
        }
        
        Result::Ok(result)
    }
    
    /// 写入全部数据
    pub fn write_all(&mut self, buf: &[u8]) -> Result<(), IoError> {
        let mut written = 0;
        while written < buf.len() {
            let count = unsafe {
                sys_write(self.fd, &buf[written], buf.len() - written)
            };
            if count <= 0 {
                return Result::Err(IoError::Other("write failed".to_string()));
            }
            written += count as usize;
        }
        Result::Ok(())
    }
    
    /// 刷新（对于文件通常无操作）
    pub fn flush(&mut self) -> Result<(), IoError> {
        unsafe { sys_fsync(self.fd); }
        Result::Ok(())
    }
    
    /// 文件大小
    pub fn metadata(&self) -> Result<FileMetadata, IoError> {
        let stat = unsafe { sys_fstat(self.fd) };
        Result::Ok(FileMetadata {
            size: stat.size,
            is_file: stat.is_file,
            is_dir: stat.is_dir,
            modified: stat.mtime,
        })
    }
}

/// 标准输入输出
pub mod stdin {
    pub fn read_line() -> Result<String, IoError> {
        let mut result = String::new();
        loop {
            let ch = unsafe { sys_read_char(0) };
            if ch == '\n' as i32 {
                break;
            }
            if ch == '\r' as i32 {
                continue;
            }
            if ch < 0 {
                break;
            }
            result.push(ch as u8 as char);
        }
        Result::Ok(result)
    }
}

pub mod stdout {
    pub fn write(s: &str) -> Result<(), IoError> {
        let bytes = s.as_bytes();
        let mut written = 0;
        while written < bytes.len() {
            let count = unsafe {
                sys_write(1, &bytes[written], bytes.len() - written)
            };
            if count <= 0 {
                return Result::Err(IoError::Other("stdout write failed".to_string()));
            }
            written += count as usize;
        }
        Result::Ok(())
    }
    
    pub fn writeln(s: &str) -> Result<(), IoError> {
        write(s)?;
        write("\n")
    }
    
    pub fn flush() -> Result<(), IoError> {
        unsafe { sys_fsync(1); }
        Result::Ok(())
    }
}

pub mod stderr {
    pub fn write(s: &str) -> Result<(), IoError> {
        let bytes = s.as_bytes();
        let mut written = 0;
        while written < bytes.len() {
            let count = unsafe {
                sys_write(2, &bytes[written], bytes.len() - written)
            };
            if count <= 0 {
                return Result::Err(IoError::Other("stderr write failed".to_string()));
            }
            written += count as usize;
        }
        Result::Ok(())
    }
    
    pub fn writeln(s: &str) -> Result<(), IoError> {
        write(s)?;
        write("\n")

/// 互斥锁（底层用 futex 或原子操作）
pub struct Mutex<T> {
    /// 0 = unlocked, 1 = locked
    state: AtomicUsize,
    data: UnsafeCell<T>,
}

/// 互斥锁守卫
pub struct MutexGuard<'a, T> {
    mutex: &'a Mutex<T>,
}

impl<T> Mutex<T> {
    pub fn new(value: T) -> Mutex<T> {
        Mutex {
            state: AtomicUsize::new(0),
            data: UnsafeCell::new(value),
        }
    }
    
    /// 加锁（阻塞直到获取）
    pub fn lock(&self) -> MutexGuard<T> {
        loop {
            // 尝试 CAS: 0 -> 1
            if self.state.compare_exchange(
                0, 1,
                Acquire, Relaxed,
            ).is_ok() {
                return MutexGuard { mutex: self };
            }
            // 自旋（生产环境应该用 futex 等待）
            spin_wait();
        }
    }
    
    /// 尝试加锁（不阻塞）
    pub fn try_lock(&self) -> Option<MutexGuard<T>> {
        if self.state.compare_exchange(
            0, 1,
            Acquire, Relaxed,
        ).is_ok() {
            Some(MutexGuard { mutex: self })
        } else {
            Option::None
        }
    }
}

impl<T> Deref for MutexGuard<'_, T> {
    type Target = T;
    
    fn deref(&self) -> &T {
        unsafe { &*self.mutex.data.get() }
    }
}

impl<T> DerefMut for MutexGuard<'_, T> {
    fn deref_mut(&mut self) -> &mut T {
        unsafe { &mut *self.mutex.data.get() }
    }
}

/// Guard 析构时自动解锁
impl<T> Drop for MutexGuard<'_, T> {
    fn drop(&mut self) {
        self.mutex.state.store(0, Release);
        // 唤醒等待者（futex_wake）
    }
}

// crates/rlyeh-std/src/sync/channel.rl

/// 有界通道发送端
pub struct Sender<T> {
    queue: Arc<LockFreeQueue<T>>,
    capacity: usize,
}

/// 有界通道接收端
pub struct Receiver<T> {
    queue: Arc<LockFreeQueue<T>>,
    capacity: usize,
}

/// 通道错误
pub enum SendError<T> {
    Disconnected(T),
    Full(T),
}

pub enum RecvError {
    Empty,
    Disconnected,
    Timeout,
}

/// 创建通道
pub fn channel<T>(capacity: usize) -> (Sender<T>, Receiver<T>) {
    let queue = Arc::new(LockFreeQueue::new(capacity));
    let sender = Sender {
        queue: queue.clone(),
        capacity,
    };
    let receiver = Receiver {
        queue,
        capacity,
    };
    (sender, receiver)
}

impl<T> Sender<T> {
    /// 发送（阻塞直到有空间或断开）
    pub fn send(&self, value: T) -> Result<(), SendError<T>> {
        loop {
            if self.queue.push(value) {
                return Result::Ok(());  // 假设我们有返回值
            }
            // 队列满，等待
            if self.queue.is_closed() {
                return Result::Err(SendError::Disconnected(value));
            }
            spin_wait();
        }
    }
    
    /// 尝试发送（不阻塞）
    pub fn try_send(&self, value: T) -> Result<(), SendError<T>> {
        if self.queue.push(value) {
            Result::Ok(())
        } else {
            Result::Err(SendError::Full(value))
        }
    }
}

impl<T> Receiver<T> {
    /// 接收（阻塞）
    pub fn recv(&self) -> Result<T, RecvError> {
        loop {
            if let Some(value) = self.queue.pop() {
                return Result::Ok(value);
            }
            if self.queue.is_closed() && self.queue.is_empty() {
                return Result::Err(RecvError::Disconnected);
            }
            spin_wait();
        }
    }
    
    /// 尝试接收（不阻塞）
    pub fn try_recv(&self) -> Result<T, RecvError> {
        if let Some(value) = self.queue.pop() {
            Result::Ok(value)
        } else if self.queue.is_closed() {
            Result::Err(RecvError::Disconnected)
        } else {
            Result::Err(RecvError::Empty)
        }
    }
}
