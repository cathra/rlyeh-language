实现 Rlyeh 的区域（Region）系统，包括：
1. **区域检查器**：验证区域内引用的安全性、转移的正确性

4. **扩容策略**：支持多种增长策略

## 核心算法

### Bump Allocator

```rust
// crates/rlyeh-region-alloc/src/lib.rs

#![warn(missing_docs)]

use std::alloc::{alloc, dealloc, Layout};
use std::ptr::NonNull;

/// 单个内存块
struct MemoryBlock {
    start: NonNull<u8>,
    end: NonNull<u8>,
    current: NonNull<u8>,  // bump pointer
    size: usize,
}

impl MemoryBlock {
    fn new(size: usize) -> Result<Self, AllocError> {
        let layout = Layout::from_size_align(size, 16).unwrap();
        let ptr = unsafe { alloc(layout) };
        if ptr.is_null() {
            return Err(AllocError::OutOfMemory);
        }
        
        let start = NonNull::new(ptr).unwrap();
        Ok(Self {
            start,
            end: unsafe { NonNull::new_unchecked(ptr.add(size)) },
            current: start,
            size,
        })
    }
    
    /// Bump allocate
    fn allocate(&mut self, size: usize, align: usize) -> Option<NonNull<u8>> {
        let current_addr = self.current.as_ptr() as usize;
        let aligned = (current_addr + align - 1) & !(align - 1);
        
        let new_current = aligned + size;
        if new_current > self.end.as_ptr() as usize {
            return None;  // 空间不足
        }
        
        let ptr = unsafe { NonNull::new_unchecked(aligned as *mut u8) };
        self.current = unsafe { NonNull::new_unchecked(new_current as *mut u8) };
        Some(ptr)
    }
    
    /// 重置（O(1)）
    fn reset(&mut self) {
        self.current = self.start;
    }
    
    /// 已用字节数
    fn used(&self) -> usize {
        self.current.as_ptr() as usize - self.start.as_ptr() as usize
    }
    
    /// 剩余字节数
    fn remaining(&self) -> usize {
        self.end.as_ptr() as usize - self.current.as_ptr() as usize
    }
}

impl Drop for MemoryBlock {
    fn drop(&mut self) {
        let layout = Layout::from_size_align(self.size, 16).unwrap();
        unsafe { dealloc(self.start.as_ptr(), layout) };
    }
}
```

### 区域结构

```rust
/// 区域（支持多块链表）
pub struct Region {
    blocks: Vec<MemoryBlock>,
    current_block: usize,
    name: Option<String>,
    
    /// 析构函数列表
    destructors: Vec<DestructorEntry>,
    
    /// 已迁移对象集合（transferred out）
    transferred: Vec<NonNull<u8>>,
    
    /// 扩容策略
    strategy: GrowthStrategy,
    
    /// 统计
    stats: RegionStats,
}

struct DestructorEntry {
    ptr: NonNull<u8>,
    drop_fn: unsafe fn(NonNull<u8>),
    size: usize,
    align: usize,
}

/// 扩容策略
pub enum GrowthStrategy {
    /// 精确大小，不允许扩容
    Exact(usize),
    
    /// 固定倍数（如 2.0 = 每次翻倍）
    Multiply(f64),
    
    /// 线性增长（每次增加固定大小）
    Linear(usize),
    
    /// 自适应（基于历史数据）
    Adaptive {
        min_size: usize,
        max_size: usize,
        growth_history: Vec<GrowthEvent>,
        allocation_samples: Vec<usize>,
    },
}

struct GrowthEvent {
    old_size: usize,
    new_size: usize,
    timestamp: std::time::Instant,
}

struct RegionStats {
    total_allocated: usize,
    total_deallocated: usize,
    allocation_count: usize,
    growth_count: usize,
    destructor_calls: usize,
}
```

## 区域检查器

```rust
// crates/rlyeh-regionck/src/lib.rs

use rlyeh_hir::*;
use thiserror::Error;

/// 区域检查器
pub struct RegionChecker {
    /// 当前活跃区域栈（支持嵌套）
    region_stack: Vec<RegionInfo>,
    errors: Vec<RegionError>,
}

struct RegionInfo {
    name: String,
    depth: usize,  // 嵌套深度
}

/// 区域错误
#[derive(Debug, Error)]
pub enum RegionError {
    #[error("reference escapes region at {line}:{col}: {detail}")]
    RegionEscape {
        detail: String,
        line: usize,
        col: usize,
    },
    
    #[error("cannot transfer reference at {line}:{col}: {detail}")]
    InvalidTransfer {
        detail: String,
        line: usize,
        col: usize,
    },
    
    #[error("region '{name}' not found at {line}:{col}")]
    RegionNotFound {
        name: String,
        line: usize,
        col: usize,
    },
    
    #[error("partial transfer not allowed at {line}:{col}: {detail}")]
    PartialTransfer {
        detail: String,
        line: usize,
        col: usize,
    },
    
    #[error("double transfer at {line}:{col}: object already transferred")]
    DoubleTransfer {
        line: usize,
        col: usize,
    },
}

impl RegionChecker {
    pub fn new() -> Self {
        Self {
            region_stack: vec![],
            errors: vec![],
        }
    }
    
    /// 检查整个程序
    pub fn check_program(&mut self, program: HirProgram) -> Result<HirProgram, Vec<RegionError>> {
        for item in &program.items {
            self.check_item(item);
        }
        
        if self.errors.is_empty() {
            Ok(program)
        } else {
            Err(self.errors.clone())
        }
    }
    
    fn check_item(&mut self, item: &HirItem) {
        match item {
            HirItem::FnDecl(fn_decl) => self.check_block(&fn_decl.body),
            HirItem::ActorDecl(actor) => self.check_actor(actor),
            _ => {}
        }
    }
    
    fn check_block(&mut self, block: &HirBlock) {
        for stmt in &block.stmts {
            self.check_stmt(stmt);
        }
        if let Some(expr) = &block.final_expr {
            self.check_expr(expr);
        }
    }
    
    fn check_stmt(&mut self, stmt: &HirStmt) {
        match stmt {
            HirStmt::Let { init, .. } => {
                self.check_expr(init);
                // 检查 init 表达式中是否有区域分配
                // 如果有，验证引用的安全性
            }
            HirStmt::Expr(e) => self.check_expr(e),
            _ => {}
        }
    }
    
    fn check_expr(&mut self, expr: &HirExpr) {
        match &expr.kind {
            HirExprKind::Region { name, body, .. } => {
                // 进入新区域
                self.region_stack.push(RegionInfo {
                    name: name.clone().unwrap_or_else(|| "<anon>".to_string()),
                    depth: self.region_stack.len(),
                });
                
                // 检查区域内代码
                self.check_block(body);
                
                // 离开区域
                self.region_stack.pop();
            }
            
            HirExprKind::InRegion { expr, region_name, .. } => {
                // 验证 region_name 存在
                if !self.region_stack.iter().any(|r| &r.name == region_name) {
                    self.errors.push(RegionError::RegionNotFound {
                        name: region_name.clone(),
                        line: expr.span.line,
                        col: expr.span.col,
                    });
                }
            }
            
            HirExprKind::Transfer { expr, region_name, .. } => {
                self.check_transfer(expr, region_name, expr.span);
            }
            
            // 递归检查子表达式
            HirExprKind::Binary { left, right, .. } => {
                self.check_expr(left);
                self.check_expr(right);
            }
            HirExprKind::Unary { operand, .. } => {
                self.check_expr(operand);
            }
            HirExprKind::Call { callee, args, .. } => {
                self.check_expr(callee);
                for arg in args {
                    self.check_expr(arg);
                }
            }
            HirExprKind::Block(block) => {
                self.check_block(block);
            }
            _ => {}
        }
    }
    
    /// 检查 transfer 的合法性
    fn check_transfer(&mut self, expr: &HirExpr, region_name: &str, span: Span) {
        // 1. 验证区域存在
        let region_idx = self.region_stack.iter().position(|r| &r.name == region_name);
        let region_idx = match region_idx {
            Some(idx) => idx,
            None => {
                self.errors.push(RegionError::RegionNotFound {
                    name: region_name.to_string(),
                    line: span.line,
                    col: span.col,
                });
                return;
            }
        };
        
        // 2. 检查 transfer 的对象是否在当前区域内
        // （不能是外层区域的对象）
        // 3. 检查不是引用（不能 transfer 引用）
        // 4. 检查没有重复 transfer
    }
}
```

```rust
        assert_eq!(*p, i as u64);

#[test]
fn test_destructor_called() {
    use std::sync::atomic::{AtomicUsize, Ordering};
    static DROP_COUNT: AtomicUsize = AtomicUsize::new(0);
    
    struct TrackDrop;
    impl Drop for TrackDrop {
        fn drop(&mut self) {
            DROP_COUNT.fetch_add(1, Ordering::SeqCst);
        }
    }
    
    let mut region = Region::new(None, GrowthStrategy::Exact(256)).unwrap();
    for _ in 0..100 {
        region.allocate(TrackDrop).unwrap();
    }
    
    assert_eq!(DROP_COUNT.load(Ordering::SeqCst), 0); // 还没销毁
    unsafe { region.destroy(); }
    assert_eq!(DROP_COUNT.load(Ordering::SeqCst), 100); // 全部析构
}

#[test]
fn test_growth_multiply() {
    let mut region = Region::new(None, GrowthStrategy::Multiply(2.0)).unwrap();
    let initial_capacity = region.stats().total_allocated;
    
    // 填满第一块
    let big = vec![0u8; initial_capacity];
    region.allocate(big).unwrap();
    
    // 再分配应该触发扩容
    let more = vec![0u8; 100];
    region.allocate(more).unwrap();
    
    assert!(region.stats().growth_count >= 1);
}
```

> 落地偏差说明：
> - `rlyeh-regionck` 未拆分 `escape.rs` / `transfer.rs`，检查逻辑合并于 `checker.rs`（区域栈 + 归属表 + 转移表）。
> - `RegionChecker` 消费 `rlyeh-typecheck::typecheck_source` 产出的 HIR；错误中的 `line`/`col` 为占位 0（HIR 节点不携带源码位置，待 Span 传播）。
> - `GrowthStrategy` 采用 struct 变体：`Exact` / `Multiply { factor }` / `Linear { chunk }` / `Adaptive { adaptive_samples, max_capacity }`。
> - `Region::new()` 无参数（默认 8B 初始块、Multiply×2），另提供 `with_initial_size` / `named`；`capacity()` = 所有块总容量（多块链表语义）。
> - `let x = expr in 'r` 的 HIR 中 `InRegion` 包裹的是初始化表达式本身，变量 `x` 的归属由 `Let` 层记录。
> - 引用逃逸（RegionEscape）完整分析依赖类型信息，MVP 保留错误变体与构造入口，待类型标注进入 HIR 后启用。
