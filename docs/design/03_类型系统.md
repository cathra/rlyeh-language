## 类型推断算法

### Hindley-Milner 风格

```rust
struct TypeChecker {
    type_env: HashMap<String, Type>,
    constraints: Vec<Constraint>,
    substitutions: HashMap<TypeVar, Type>,
}

impl TypeChecker {
    fn infer_expr(&mut self, expr: &Expr) -> Type {
        match expr {
            Expr::Literal(Literal::Integer(n)) => {
                // 整数默认 i32，但可向下转型
                Type::I32
            }
            Expr::Literal(Literal::Float(f)) => Type::F64,
            Expr::Ident(name) => self.lookup_var(name),
            
            Expr::Binary(lhs, op, rhs) => {
                let t1 = self.infer_expr(lhs);
                let t2 = self.infer_expr(rhs);
                self.unify(&t1, &t2);
                self.check_binop_type(op, &t1)
            }
            
            // 比较链类型检查（核心）
            Expr::ComparisonChain(ops, exprs) => {
                // 1. 检查所有表达式类型一致或可统一
                let types: Vec<Type> = exprs.iter()
                    .map(|e| self.infer_expr(e))
                    .collect();
                
                for i in 0..types.len() - 1 {
                    self.unify(&types[i], &types[i+1]);
                }
                
                // 2. 检查类型实现了 PartialOrd
                let unified = self.substitute(&types[0]);
                if !self.implements(&unified, "PartialOrd") {
                    self.error("type does not implement PartialOrd");
                }
                
                // 3. 检查比较链方向一致性
                self.check_chain_direction(ops);
                
                Type::Bool
            }
            
            // in 表达式
            Expr::In(expr, patterns) => {
                let expr_type = self.infer_expr(expr);
                
                for pattern in patterns {
                    let pat_type = self.infer_pattern_type(pattern);
                    self.unify(&expr_type, &pat_type);
                    
                    // 检查 PartialEq
                    if !self.implements(&expr_type, "PartialEq") {
                        self.error("type must implement PartialEq for 'in'");
                    }
                }
                
                Type::Bool
            }
            
            _ => todo!(),
        }
    }
}
```

## 比较链方向检查

```rust
enum ChainDirection {
    Forward,   // 0 < i < 10  (区间内)
    Backward,  // 0 > i > 10  (区间外)
    Mixed,     // 0 < i > 10  (错误！)
}

fn check_chain_direction(&self, ops: &[CompOp]) -> ChainDirection {
    let first_is_less = matches!(ops[0], CompOp::Lt | CompOp::Le);
    
    for op in &ops[1..] {
        let is_less = matches!(op, CompOp::Lt | CompOp::Le);
        if is_less != first_is_less {
            self.error("comparison chain direction is inconsistent");
        }
    }
    
    if first_is_less { Forward } else { Backward }
}
```
